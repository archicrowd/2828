Blatt 4 · Simulation

Zwei Zukünfte, eine Entscheidung

Vier Kenngrößen von 2010 bis 2035: die Entwicklung bis heute und ab 2028 zwei Pfade – weiter wie bisher oder mit 28:28.

Verlauf bis 2025 ohne 28:28 mit 28:28 (vorsichtige Annahmen)

Pfade ab 2028 sind Modellrechnungen, keine Prognosen. Schulden: Quote abzüglich Fondsreserve. Ungleichheit: Modellschätzung, belastbar ist eine Größenordnung von 2 bis 5 Gini-Punkten. CO₂-Pfad aus der Vorfassung übernommen.

Trägt der Haushalt das?

Ja – in allen drei Rechenwegen ohne ein einziges Jahr mit Finanzierungslücke. Vor der ersten Auszahlung entsteht ein Polster; danach bleiben jedes Jahr bis zu 1 % des BIP für flexible Investitionen, weitere 2 % liegen als Krisenreserve bereit.

Auch im Rezessionstest wird die Dividende nicht gekürzt. Nur die jährliche Anpassung pausiert; ihr Wert sinkt dann zeitweise um bis zu 9 %.

Blatt 5 · Sicherungen

Warum der Haushalt nicht kippen kann

Ein Entwurf ist so gut wie seine Sicherungen. Fünf Regeln entscheiden – nicht Wahlkämpfe.

Einnahmen zuerst.

Konzerne, Plattformen, Kapital, Erbschaften, große Vermögen und Boden werden mehrere Monate vor der ersten Auszahlung nach neuem Recht besteuert. Die Dividende startet auf einem Polster, nie auf Kredit.

Formel statt Kalender.

Die Dividende beträgt die Hälfte des amtlichen Existenzminimums. Sie steigt in Schritten bis 60 %, sobald die Krisenreserve gefüllt und die Investitionen gesichert sind.

Nie kürzen – nur pausieren.

Wird die Reserve angegriffen, setzt nur die jährliche Anpassung aus. Nachgeholt wird ausschließlich aus Überschüssen.

Reserve und Investitionsfonds.

2 % des BIP liegen als Krisenreserve bereit; darüber hinaus fließt jedes Jahr bis zu 1 % des BIP in Schulen, Netze und Schienen. Der Staat bleibt handlungsfähig, ohne neue Schulden.

Fiskalrat, offene Zahlen, fester Korridor.

Ein unabhängiger Rat prüft jährlich alle Annahmen; die Ist-Zahlen sind monatlich öffentlich. Als letzte Sicherung darf der Einheitssatz nur regelgebunden zwischen 25 und 30 % gleiten.

AN_KINDERLOS = 0.006

def est_2026(zve):
    zve = math.floor(max(0, zve))
    if zve <= 12348: return 0.0
    if zve <= 17799: y = (zve-12348)/1e4; return math.floor((914.51*y+1400)*y)
    if zve <= 69878: z = (zve-17799)/1e4; return math.floor((173.10*z+2397)*z+1034.87)
    if zve <= 277825: return math.floor(0.42*zve-11135.63)
    return math.floor(0.45*zve-19470.38)

def soli_2026(est, split=False):
    fg = SOLI_FG*(2 if split else 1)
    if est <= fg: return 0.0
    return min(0.055*est, 0.119*(est-fg))

def sv_heute(b, kinderlos=True):
    kvpv = (AN_KV_PV + (AN_KINDERLOS if kinderlos else 0))*min(b, BBG_KV)
    return kvpv + AN_RV_AV*min(b, BBG_RV)

def netto_heute(b, kinder=0, partner=False):
    sv = sv_heute(b, kinderlos=(kinder == 0))
    zve = max(0, b - sv - WK_PAUSCH*(b > 0) - SA_PAUSCH*(2 if partner else 1))
    est = 2*est_2026(zve/2) if partner else est_2026(zve)
    soli = soli_2026(est, partner)
    return dict(netto=b - sv - est - soli + kinder*KINDERGELD*12, est=est+soli, sv=sv)

def est_2828(x, saetze=TARIF_SAETZE, grenzen=TARIF_GRENZEN):
    s, prev = 0.0, 0.0
    for g, r in zip(grenzen, saetze):
        if x > prev: s += (min(x, g)-prev)*r
        prev = g
        if x <= g: break
    return s

def netto_2828(b, kinder=0, partner=False, anteil=50):
    dd = DD_EUR[anteil]*12
    sv = SV_AN[anteil]*b                          # keine Bemessungsgrenzen
    est = est_2828(b + dd)                        # DD steuerpflichtig, Individualbesteuerung
    kind = kinder*DD_EUR[anteil]*KIND_ANTEIL*12   # Kinder-DD steuerfrei (Annahme)
    p = 0.0
    if partner: p = dd - est_2828(dd)             # Partner: eigene DD, eigene Veranlagung
    return dict(netto=b - sv - est + dd + kind + p, est=est, sv=sv, dd=dd + kind + p)

def r_eff_mwst(stufen=MWST_STUFEN):
    return sum(w*s for w, s in zip(MWST_GEWICHTE, stufen))/100
R_EFF_HEUTE = 0.165                               # Projektannahme effektiver Satz heute

def konsumquote(n): return .95 if n < 25000 else .85 if n < 50000 else .75 if n < 80000 else .60
def mwst_mehrbelastung(netto):
    r = r_eff_mwst()
    return netto*konsumquote(netto)*(r/(1+r) - R_EFF_HEUTE/(1+R_EFF_HEUTE))

# Kalibrierung wie Projekt: 25er-Tarif + DD 6.000 = 284 Mrd.
K_EST = 284/sum(mio*est_2828(b+6000, S25, G25)/1000 for mio, b in VERT)

SZ = {

def est_aufkommen(anteil, eti):

    for mio, b in VERT:
        m0, m1 = _grenz(b+dd, S25, G25), _grenz(b+dd, TARIF_SAETZE, TARIF_GRENZEN)

def mwst_aufkommen(leak):

    return 350 + (brutto*r1/(1+r1) - 350)*leak

def einnahmen(anteil, sc, detail=False):

    d = {
     'Einkommensteuer (inkl. steuerpfl. DD)': est_aufkommen(anteil, sc['eti']),

     'Erbschaft 28 %':                        33*(t/.25)*math.exp(-(L-25)/100),
     'Vermögen-Mindeststeuer (anrechenbar)':  22*(1-0.02*(L-25)),
     'CO₂-Abgabe 100 €/t (netto)':            55,
     'Sonstige Steuern':                      165,

def simuliere(sc, mit_bausteinen=True, jahre=20):
    g_e = sc['g_e'] - sc['drag']*(L-25); g_a = sc['g_a'] - 0.5*sc['drag']*(L-25)
    E = {a: einnahmen(a, sc) for a in STUFEN}
    F, i, idx, idx_voll, letzter_schritt, rows = STARTPUFFER, 0, 1.0, 1.0, -99, []
    for t in range(jahre):
        bip = BIP0*(1+g_e)**t
        if t > 0: idx_voll *= 1+sc['dd_idx']
        def saldo(a, ix):
            e = (E[a] + (bausteine(t)*sc['hc'] if mit_bausteinen else 0))*(1+g_e)**t*sc['schock'].get(t,1.0) \
                + (sc['verh'] if t >= 2 else 0)
            x = AUSG[a]; aus = (x['basis']+x['zusch']+x['haerte'])*(1+g_a)**t + dd_kosten(a)*ix + sc['schock_a'].get(t,0)
            return e - aus
        # Regel 3: jaehrliche Anpassung an das Existenzminimum, solange der Puffer nicht angegriffen ist;
        #          ausgesetzte Anpassungen werden nur aus Ueberschuessen nachgeholt

        # Regel 2: Aufstieg 50->55->60 %, fruehestens Jahr 3, Abstand >= 2 Jahre
        rest = max(F - PUFFER*bip, 0)
        if i < 2 and t >= 2 and t - letzter_schritt >= 2 and F >= PUFFER*bip \
           and saldo(STUFEN[i+1], idx) + rest/10 >= INV*bip:
            i += 1; letzter_schritt = t
        S = saldo(STUFEN[i], idx); F += S
        luecke = F < 0
        inv = min(max(F - PUFFER*bip, 0), INV*bip); F -= inv
        rows.append(dict(jahr=START+t, t=t+1, anteil=STUFEN[i], dd=DD_EUR[STUFEN[i]]*idx, dd_2026=DD_EUR[STUFEN[i]]*idx/idx_voll, dd_real=idx/idx_voll,
                         saldo=S, inv=inv, ziel=INV*bip, fonds=F, luecke=luecke, bip=bip))
    return rows

/* 28:28 – Rechnerlogik (Recht 2026 vs. 28:28). Alle Beträge EUR/Jahr, Preise 2026.
   Quelle heute: § 32a EStG 2026, SV-Rechengrößen 2026, Soli-Freigrenze 2026, Kindergeld 259 €. */
const P = {
  exmin: 12348, dd: {50: 515, 55: 566, 60: 617}, kindAnteil: 0.5,
  tarifGrenzen: [12348, 25000, 40000, 55000, 70000, Infinity],
  tarifSaetze:  [0, .09, .16, .21, .26, .28],
  svAN: {50: .14, 55: .14, 60: .09},
  bbgKV: 69750, bbgRV: 101400, anKVPV: .1055, anKinderlos: .006, anRVAV: .106,
  kindergeld: 259, wk: 1230, sa: 36, soliFG: 20350
};
function est2026(zve){ zve=Math.floor(Math.max(0,zve));
  if(zve<=12348) return 0;
  if(zve<=17799){const y=(zve-12348)/1e4; return Math.floor((914.51*y+1400)*y);}
  if(zve<=69878){const z=(zve-17799)/1e4; return Math.floor((173.10*z+2397)*z+1034.87);}
  if(zve<=277825) return Math.floor(.42*zve-11135.63);
  return Math.floor(.45*zve-19470.38); }
function soli2026(e,split){const fg=P.soliFG*(split?2:1); return e<=fg?0:Math.min(.055*e,.119*(e-fg));}
function heute(b,kinder,partner){
  const sv=(P.anKVPV+(kinder===0?P.anKinderlos:0))*Math.min(b,P.bbgKV)+P.anRVAV*Math.min(b,P.bbgRV);
  const zve=Math.max(0,b-sv-(b>0?P.wk:0)-P.sa*(partner?2:1));
  const e=partner?2*est2026(zve/2):est2026(zve); const s=soli2026(e,partner);
  return {netto:b-sv-e-s+kinder*P.kindergeld*12, steuer:e+s, sv:sv};
}
function est2828(x){let s=0,prev=0;
  for(let i=0;i<P.tarifSaetze.length;i++){const g=P.tarifGrenzen[i];
    if(x>prev) s+=(Math.min(x,g)-prev)*P.tarifSaetze[i]; prev=g; if(x<=g) break;}
  return s;}
